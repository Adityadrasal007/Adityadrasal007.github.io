"use client"; 
 
import * as React from "react"; 
import { motion } from "framer-motion"; 
 
export interface ScrollReelCertification { 
  title: string; 
  issuer: string; 
  image: string; 
  url: string; 
} 
 
export interface ScrollReelCertificationsProps { 
  certifications: ScrollReelCertification[]; 
  charStaggerMs?: number; 
  className?: string; 
} 
 
// 1. UPDATED DIMENSIONS: Landscape format for certificates 
const CELL_W = 280;  
const CELL_H = 200;  
const GAP = 12; 
const STEP = 3 * (CELL_H + GAP); 
 
const EXIT_MS = 240; 
const SLIDE_MS = 800; 
const EASE_INOUT = "cubic-bezier(0.65,0,0.35,1)"; 
 
const TITLE_CLASSES = 
  "m-0 text-lg font-medium leading-[1.3] tracking-[-0.02em] text-white sm:text-[22px]"; 
const ISSUER_CLASSES = 
  "m-0 text-sm font-medium leading-[1.3] text-blue-400 hover:text-blue-300 transition-colors inline-flex items-center gap-1 cursor-pointer"; 
 
const FEATURED_SHADOW = 
  "0 10px 30px -10px rgba(0,0,0,0.8), inset 0 1px 0 rgba(255,255,255,0.1)"; 
 
function cn(...classes: Array<string | false | null | undefined>) { 
  return classes.filter(Boolean).join(" "); 
} 
 
/* Background Placeholder Cells */ 
function Cell() { 
  return ( 
    <div 
      aria-hidden="true" 
      className="shrink-0 rounded-xl border border-gray-800/50 bg-gradient-to-b from-gray-900/40 to-black blur-[3px]" 
      style={{ width: CELL_W, height: CELL_H }} 
    /> 
  ); 
} 
 
/* 2 & 3. UPDATED FEATURED COMPONENT: Full visibility + Floating Animation */ 
function Featured({ src, alt }: { src: string; alt?: string }) { 
  return ( 
    <motion.div 
      // Continuous floating animation 
      animate={{ y: [0, -8, 0] }} 
      transition={{  
        duration: 4,  
        repeat: Infinity,  
        ease: "easeInOut"  
      }} 
      className="relative shrink-0 overflow-hidden rounded-xl bg-[#050505] ring-1 ring-gray-700/50 group" 
      style={{ width: CELL_W, height: CELL_H, boxShadow: FEATURED_SHADOW }} 
    > 
      <img 
        src={src} 
        alt={alt ?? "Certification"} 
        loading="lazy" 
        // object-contain ensures the whole certificate is visible without cropping 
        className="absolute inset-0 h-full w-full object-contain p-2" 
      /> 
        
      {/* Animated Light Sheen/Glare over the certificate */} 
      <motion.div 
        aria-hidden="true" 
        animate={{  
          x: ["-100%", "200%"], 
        }} 
        transition={{ 
          duration: 3, 
          repeat: Infinity, 
          repeatDelay: 5, // Pauses for 5 seconds between glares 
          ease: "linear" 
        }} 
        className="pointer-events-none absolute inset-0 z-10 w-1/2 -skew-x-12 opacity-20" 
        style={{ 
          background: "linear-gradient(90deg, transparent, rgba(255,255,255,0.8), transparent)", 
        }} 
      /> 
    </motion.div> 
  ); 
} 
 
function Chars({ 
  text, 
  startIndex, 
  staggerMs, 
}: { 
  text: string; 
  startIndex: number; 
  staggerMs: number; 
}) { 
  let idx = startIndex; 
  const words = text.split(" "); 
  return ( 
    <> 
      {words.map((word, wi) => { 
        const wordSpan = ( 
          <span className="inline-block whitespace-nowrap"> 
            {Array.from(word).map((ch, ci) => { 
              const delay = idx * staggerMs; 
              idx++; 
              return ( 
                <span 
                  key={ci} 
                  className="scroll-reel-char inline-block will-change-transform" 
                  style={{  
                    animationDelay: `${delay}ms`, 
                    animation: `slideUpFade 0.4s cubic-bezier(0.16, 1, 0.3, 1) forwards`, 
                    opacity: 0, 
                    transform: 'translateY(10px)' 
                  }} 
                > 
                  {ch} 
                </span> 
              ); 
            })} 
          </span> 
        ); 
        if (wi < words.length - 1) idx++; 
        return ( 
          <React.Fragment key={wi}> 
            {wordSpan} 
            {wi < words.length - 1 ? " " : null} 
          </React.Fragment> 
        ); 
      })} 
    </> 
  ); 
} 
 
export function ScrollReelCertifications({ 
  certifications = myCertifications, // Using the updated array as the default fallback
  charStaggerMs = 8, 
  className, 
}: ScrollReelCertificationsProps) { 
  const [index, setIndex] = React.useState(0); 
  const [displayIndex, setDisplayIndex] = React.useState(0); 
  const [exiting, setExiting] = React.useState(false); 
  const [mounted, setMounted] = React.useState(false); 
  const animating = React.useRef(false); 
  const timeouts = React.useRef<ReturnType<typeof setTimeout>[]>([]); 
 
  const count = certifications.length; 
 
  React.useEffect(() => { 
    const raf = requestAnimationFrame(() => 
      requestAnimationFrame(() => setMounted(true)) 
    ); 
    return () => { 
      cancelAnimationFrame(raf); 
      timeouts.current.forEach(clearTimeout); 
    }; 
  }, []); 
 
  const paginate = React.useCallback( 
    (dir: 1 | -1) => { 
      if (animating.current) return; 
      const next = index + dir; 
      if (next < 0 || next >= count) return; 
      animating.current = true; 
 
      setIndex(next); 
      setExiting(true); 
 
      timeouts.current.push( 
        setTimeout(() => { 
          setDisplayIndex(next); 
          setExiting(false); 
        }, EXIT_MS) 
      ); 
      timeouts.current.push( 
        setTimeout(() => { 
          animating.current = false; 
        }, SLIDE_MS) 
      ); 
    }, 
    [index, count] 
  ); 
 
  const onKeyDown = (e: React.KeyboardEvent) => { 
    if (e.key === "ArrowRight") { 
      e.preventDefault(); 
      paginate(1); 
    } 
    if (e.key === "ArrowLeft") { 
      e.preventDefault(); 
      paginate(-1); 
    } 
  }; 
 
  const middleItems = React.useMemo(() => { 
    const items: Array<{ type: "cell" } | { type: "featured"; i: number }> = []; 
    for (let i = 0; i < 3; i++) items.push({ type: "cell" }); 
    certifications.forEach((_, i) => { 
      items.push({ type: "featured", i }); 
      if (i < count - 1) { 
        items.push({ type: "cell" }, { type: "cell" }); 
      } 
    }); 
    for (let i = 0; i < 3; i++) items.push({ type: "cell" }); 
    return items; 
  }, [certifications, count]); 
 
  const sideCellCount = 4 + 2 * count; 
  const centerIdx = (count - 1) / 2; 
  const middleY = (centerIdx - index) * STEP; 
  const sideY = -middleY; 
 
  const colStyle = (y: number): React.CSSProperties => ({ 
    transform: `translateY(${y}px)`, 
    transition: mounted ? `transform ${SLIDE_MS}ms ${EASE_INOUT}` : "none", 
  }); 
 
  const current = certifications[displayIndex]; 
 
  return ( 
    <div 
      role="region" 
      aria-roledescription="carousel" 
      aria-label="Certifications" 
      tabIndex={0} 
      onKeyDown={onKeyDown} 
      className={cn( 
        "relative flex w-full max-w-[1100px] flex-col items-stretch gap-4 overflow-hidden rounded-2xl border border-gray-800 bg-[#0A0A0A] outline-none focus-visible:ring-2 focus-visible:ring-gray-600 md:min-h-[360px] md:flex-row", 
        className 
      )} 
    > 
      {/* Reel section */} 
      <div 
        aria-hidden="true" 
        className="relative h-72 w-full shrink-0 self-stretch overflow-hidden md:h-auto md:w-[480px]" 
        style={{ 
          WebkitMaskImage: 
            "linear-gradient(to right, transparent 0%, black 14%, black 86%, transparent 100%), linear-gradient(to bottom, transparent 0%, black 10%, black 90%, transparent 100%)", 
          maskImage: 
            "linear-gradient(to right, transparent 0%, black 14%, black 86%, transparent 100%), linear-gradient(to bottom, transparent 0%, black 10%, black 90%, transparent 100%)", 
          WebkitMaskComposite: "source-in", 
          maskComposite: "intersect", 
        }} 
      > 
        <div className="absolute inset-0 flex items-center justify-center gap-3"> 
          {/* Left column */} 
          <div 
            className="flex shrink-0 flex-col gap-3 will-change-transform motion-reduce:[transition:none!important]" 
            style={colStyle(sideY)} 
          > 
            {Array.from({ length: sideCellCount }).map((_, i) => ( 
              <Cell key={`l-${i}`} /> 
            ))} 
          </div> 
 
          {/* Middle column */} 
          <div 
            className="flex shrink-0 flex-col gap-3 will-change-transform motion-reduce:[transition:none!important]" 
            style={colStyle(middleY)} 
          > 
            {middleItems.map((item, i) => 
              item.type === "featured" ? ( 
                <Featured 
                  key={`m-${i}`} 
                  src={certifications[item.i].image} 
                /> 
              ) : ( 
                <Cell key={`mc-${i}`} /> 
              ) 
            )} 
          </div> 
 
          {/* Right column */} 
          <div 
            className="flex shrink-0 flex-col gap-3 will-change-transform motion-reduce:[transition:none!important]" 
            style={colStyle(sideY)} 
          > 
            {Array.from({ length: sideCellCount }).map((_, i) => ( 
              <Cell key={`r-${i}`} /> 
            ))} 
          </div> 
        </div> 
      </div> 
 
      {/* Content section */} 
      <div className="flex min-w-0 flex-1 flex-col justify-between self-stretch px-6 py-8 md:py-12 bg-[#111113]"> 
        <div className="flex flex-col gap-[12px]"> 
          {/* Certificate Icon */} 
          <svg  
            className="block h-10 w-10 text-gray-500 mb-4"  
            viewBox="0 0 24 24"  
            fill="none"  
            stroke="currentColor"  
            strokeWidth="1.5"  
            strokeLinecap="round"  
            strokeLinejoin="round" 
          > 
            <path d="M12 15V3m0 12l-4-4m4 4l4-4M2 17l.621 2.485A2 2 0 0 0 4.561 21h14.878a2 2 0 0 0 1.94-1.515L22 17"></path> 
          </svg> 
 
          {/* Text stage */} 
          <div 
            className="relative w-full max-w-[450px] overflow-hidden" 
            aria-live="polite" 
          > 
            <div 
              aria-hidden="true" 
              className="invisible flex min-h-[120px] flex-col gap-[16px]" 
            > 
              <p className={TITLE_CLASSES}>{current.title}</p> 
              <p className={ISSUER_CLASSES}>{current.issuer}</p> 
            </div> 
              
            <div 
              key={displayIndex} 
              className={cn( 
                "absolute inset-x-0 top-0 flex flex-col gap-[16px] will-change-[transform,opacity] transition-opacity duration-200", 
                exiting ? "opacity-0" : "opacity-100" 
              )} 
            > 
              <p className={TITLE_CLASSES}> 
                <Chars 
                  text={current.title} 
                  startIndex={0} 
                  staggerMs={charStaggerMs} 
                /> 
              </p> 
                
              <a  
                href={current.url}  
                target="_blank"  
                rel="noreferrer" 
                className={ISSUER_CLASSES} 
                style={{ animationDelay: `${current.title.length * charStaggerMs}ms` }} 
              > 
                <Chars 
                  text={current.issuer} 
                  startIndex={current.title.length + 5} 
                  staggerMs={charStaggerMs} 
                /> 
                <svg className="w-3.5 h-3.5 ml-1 mt-0.5 opacity-70" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"> 
                  <path d="M18 13v6a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V8a2 2 0 0 1 2-2h6M15 3h6v6M10 14L21 3" /> 
                </svg> 
              </a> 
            </div> 
          </div> 
        </div> 
 
        {/* Controls */} 
        <div className="mt-8 flex items-center gap-3 md:mt-0"> 
          <button 
            type="button" 
            onClick={() => paginate(-1)} 
            disabled={index === 0} 
            aria-label="Previous certification" 
            className="grid h-10 w-10 cursor-pointer place-items-center rounded-full border border-gray-700 bg-transparent p-0 text-white transition-all hover:enabled:bg-gray-800 hover:enabled:scale-[1.05] active:enabled:scale-[0.95] disabled:cursor-default disabled:opacity-30" 
          > 
            <svg className="h-4 w-4" viewBox="0 0 12 12" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round"> 
              <path d="M7.5 2.5 3.5 6l4 3.5" /> 
            </svg> 
          </button> 
          <button 
            type="button" 
            onClick={() => paginate(1)} 
            disabled={index === count - 1} 
            aria-label="Next certification" 
            className="grid h-10 w-10 cursor-pointer place-items-center rounded-full border border-gray-700 bg-transparent p-0 text-white transition-all hover:enabled:bg-gray-800 hover:enabled:scale-[1.05] active:enabled:scale-[0.95] disabled:cursor-default disabled:opacity-30" 
          > 
            <svg className="h-4 w-4" viewBox="0 0 12 12" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round"> 
              <path d="m4.5 2.5 4 3.5-4 3.5" /> 
            </svg> 
          </button> 
        </div> 
      </div> 
        
      {/* Required CSS for the text animation */} 
      <style dangerouslySetInnerHTML={{__html: ` 
        @keyframes slideUpFade { 
          from { opacity: 0; transform: translateY(10px); } 
          to { opacity: 1; transform: translateY(0); } 
        } 
      `}} /> 
    </div> 
  ); 
} 
 
export default ScrollReelCertifications;
